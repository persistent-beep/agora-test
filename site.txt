function debounce(func, delay) {
    let timeout;
    return (...args) => {
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(this, args), delay);
    };
}

// [GLARE] Детерминированное разрешение конфликта «обе стороны послали
// оффер одновременно» алфавитным порядком ролей. Роли уникальны, обе
// стороны вычисляют одно и то же без переговоров:
//   моя роль > роли собеседника → я impolite (никогда не уступаю),
//   иначе → я polite (откатываю свой оффер и принимаю чужой).
// Ровно одна сторона уступает — всегда сходится.
function getMyRoleId() {
    const session = JSON.parse(localStorage.getItem("agora_session") || "{}");
    return (session.role || userRole || "").toLowerCase();
}

function isImpolitePeer() {
    const me = getMyRoleId();
    const them = (currentCallTarget || "").toLowerCase();
    if (!me || !them) return false; // не знаем роль — считаем себя polite
    return me > them;
}

// [CALL_ID] Сообщение чужого/прошлого звонка? Если у сообщения и у нас
// есть call_id и они разные — игнор. Если call_id нет (легаси-путь,
// например сервер перезапускался и потерял реестр) — пропускаем,
// дальше сработает фильтр по msg.from (вторая линия обороны).
function isStaleCallMsg(msg) {
    if (msg.call_id && activeCallId && msg.call_id !== activeCallId) {
        console.log("[Signal] Сообщение чужого звонка, игнорируем:", msg.type);
        return true;
    }
    return false;
}

// Таймаут безответности: если за 60с звонок не приняли — сбрасываем
function startRingTimeout() {
    clearRingTimeout();
    ringTimeout = setTimeout(() => {
        ringTimeout = null;
        if (callState === "RINGING_OUT") {
            console.log("[Call] Нет ответа за 60с — сбрасываем звонок");
            hangUp();
            updateCallUI("CALL", "btn-green", "NO ANSWER", "#ff4a4a", false);
        }
    }, RING_TIMEOUT_MS);
}

function clearRingTimeout() {
    if (ringTimeout) {
        clearTimeout(ringTimeout);
        ringTimeout = null;
    }
}

async function fetchIceServers() {
    if (iceConfig.length > 0) return iceConfig;
    try {
        // [FIX SECURITY] Передаём токен: без него сервер теперь отдаёт
        // только публичный STUN, а не TURN-кредитки (см. main.py /ice-servers).
        const session = JSON.parse(
            localStorage.getItem("agora_session") || "{}",
        );
        const tokenParam = session.token
            ? `&token=${encodeURIComponent(session.token)}`
            : "";
        const response = await fetch(
            `${API_URL}/ice-servers?role=${encodeURIComponent(userRole)}${tokenParam}`,
        );
        const data = await response.json();
        iceConfig = data.iceServers || [{ urls: "stun:stun.l.google.com:19302" }];
    } catch (e) {
        console.error("Ошибка ICE, используем резервный:", e);
        iceConfig = [{ urls: "stun:stun.l.google.com:19302" }];
    }
    return iceConfig;
}

function urlBase64ToUint8Array(base64String) {
    const padding = "=".repeat((4 - base64String.length % 4) % 4);
    const base64 = (base64String + padding).replace(/-/g, "+").replace(
        /_/g,
        "/",
    );
    const rawData = window.atob(base64);
    const outputArray = new Uint8Array(rawData.length);
    for (let i = 0; i < rawData.length; ++i) {
        outputArray[i] = rawData.charCodeAt(i);
    }
    return outputArray;
}

async function subscribeToPush(token) {
    // Защита от отсутствия токена или гостевого входа
    if (!token || token === "guest") {
        console.warn(
            "[Push] Пропускаем подписку (гостевой режим или нет токена)",
        );
        return;
    }
    if (!("serviceWorker" in navigator) || !("PushManager" in window)) {
        console.warn("[Push] Браузер не поддерживает Push");
        return;
    }

    try {
        const registration = await navigator.serviceWorker.ready;

        const existing = await registration.pushManager.getSubscription();
        if (existing) await existing.unsubscribe();

        const permission = await Notification.requestPermission();
        if (permission !== "granted") {
            console.warn("[Push] Уведомления запрещены пользователем");
            return;
        }

        const subscription = await registration.pushManager.subscribe({
            userVisibleOnly: true,
            applicationServerKey: urlBase64ToUint8Array(PUBLIC_VAPID_KEY),
        });

        const response = await fetch(
            `${API_URL}/push/subscribe?token=${encodeURIComponent(token)}`,
            {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ subscription }),
            },
        );

        if (response.ok) {
            console.log("[Push] Успешно подписан на звонки");
        } else {
            console.error(
                "[Push] Ошибка сохранения подписки, статус:",
                response.status,
            );
        }
    } catch (error) {
        console.error("[Push] Ошибка подписки:", error);
    }
}

async function handleLogout() {
    if (signalingSocket) {
        // [FIX RC-5] Гасим обработчики ДО close(), чтобы onclose не запустил
        // автопереподключение перед location.reload().
        signalingSocket.onclose = null;
        signalingSocket.onerror = null;
        signalingSocket.close();
    }

    // --- ОТПИСКА ОТ PUSH ПРИ ВЫХОДЕ ---
    if ("serviceWorker" in navigator) {
        try {
            const registration = await navigator.serviceWorker.ready;
            const subscription = await registration.pushManager
                .getSubscription();

            if (subscription) {
                const session = JSON.parse(
                    localStorage.getItem("agora_session") || "{}",
                );
                if (session.token) {
                    await fetch(
                        `${API_URL}/push/unsubscribe?token=${
                            encodeURIComponent(session.token)
                        }`,
                        {
                            method: "POST",
                            headers: { "Content-Type": "application/json" },
                            body: JSON.stringify({
                                endpoint: subscription.endpoint,
                            }),
                        },
                    );
                }
                await subscription.unsubscribe();
                console.log(
                    "[Push] Успешно деактивировали пуши для этого устройства.",
                );
            }
        } catch (e) {
            console.error("[Push] Ошибка при деактивации пушей:", e);
        }
    }
    // ----------------------------------

    localStorage.removeItem("agora_session");
    location.reload();
}

async function setupAudio() {
    try {
        localStream = await navigator.mediaDevices.getUserMedia({
            audio: {
                echoCancellation: true,
                noiseSuppression: true,
                autoGainControl: true,
            },
        });
        audioContext = new (window.AudioContext || window.webkitAudioContext)();
        localAnalyser = audioContext.createAnalyser();
        localAnalyser.fftSize = 256;
        localDataArray = new Uint8Array(localAnalyser.frequencyBinCount);
        audioContext.createMediaStreamSource(localStream).connect(
            localAnalyser,
        );
    } catch (err) {
        console.error("Mic access denied:", err);
        localDataArray = new Uint8Array(128); // Пустой массив для тишины
    }
}

        // 1. ЕСЛИ СВЯЗЬ ВОССТАНОВИЛАСЬ
        if (
            newPc.iceConnectionState === "connected" ||
            newPc.iceConnectionState === "completed"
        ) {
            if (reconnectTimeout) {
                clearTimeout(reconnectTimeout); // Выключаем таймер смерти
                reconnectTimeout = null;
                updateCallUI(
                    "END CALL",
                    "btn-red",
                    "CONNECTED",
                    "#4aff4a",
                    false,
                );
                console.log("[ICE] Переподключение успешно завершено!");
            }
        }

        // 2. ЕСЛИ СВЯЗЬ РАЗОРВАЛАСЬ (Смена сети, метро, сел телефон)
        if (
            newPc.iceConnectionState === "disconnected" ||
            newPc.iceConnectionState === "failed"
        ) {
            if (callState === "CONNECTED") {
                console.log(
                    "[ICE] Смена сети или обрыв! Запуск ICE Restart...",
                );
                // [FIX UX] Кнопка остается АКТИВНОЙ: пользователь должен иметь
                // возможность самому сбросить зависающий звонок, а не ждать
                // 40 секунд заблокированной кнопкой.
                updateCallUI(
                    "END CALL",
                    "btn-vosklet",
                    "RECONNECTING",
                    "#ff9900",
                    false,
                );

                // --- ЗАПУСКАЕМ ТАЙМЕР СМЕРТИ ЗВОНКА (40 секунд) ---
                if (!reconnectTimeout) {
                    reconnectTimeout = setTimeout(() => {
                        console.log(
                            "[ICE] Время на переподключение вышло. Сбрасываем звонок.",
                        );
                        reconnectTimeout = null;
                        hangUp();
                    }, 40000);
                }

                try {
                    const offer = await newPc.createOffer({ iceRestart: true });
                    // [FIX RC-9] Пока собирали оффер, звонок могли сбросить
                    if (pc !== newPc || callState !== "CONNECTED") return;
                    await newPc.setLocalDescription(offer);
                    if (pc !== newPc || callState !== "CONNECTED") return;

                    if (signalingSocket?.readyState === WebSocket.OPEN) {
                        signalingSocket.send(JSON.stringify({
                            type: "offer",
                            target: currentCallTarget,
                            sdp: newPc.localDescription.sdp,
                        }));
                    }
                } catch (err) {
                    console.error("Ошибка ICE Restart:", err);
                    hangUp();
                }
            } else {
                hangUp();
            }
        }
    };
    // +===== DIAGNOSTICS ends
    newPc.onicecandidate = (event) => {
        if (event.candidate) {
            console.log(
                `[ICE] кандидат: тип=${event.candidate.type}, протокол=${event.candidate.protocol}, адрес=${event.candidate.address}`,
            );
        } else {
            console.log(
                `[ICE] кандидаты закончились (end-of-candidates) через ${
                    ((performance.now() - startTime) / 1000).toFixed(1)
                }s)`,
            );
        }
        if (event.candidate && signalingSocket?.readyState === WebSocket.OPEN) {
            signalingSocket.send(JSON.stringify({
                type: "candidate",
                target: currentCallTarget,
                candidate: event.candidate,
            }));
        }
    };

    newPc.ontrack = (event) => {
        if (remoteAudioElement) remoteAudioElement.srcObject = null;
        remoteAudioElement = new Audio();
        remoteAudioElement.srcObject = event.streams[0];
        remoteAudioElement.autoplay = true;
        remoteAudioElement.play().catch((e) => {
            console.log("Autoplay blocked", e);
            alert("Нажмите 'ОК', чтобы разрешить звук в браузере");
        });

        if (audioContext && audioContext.state !== "closed") {
            remoteAnalyser = audioContext.createAnalyser();
            remoteAnalyser.fftSize = 256;
            remoteDataArray = new Uint8Array(remoteAnalyser.frequencyBinCount);
            audioContext.createMediaStreamSource(event.streams[0]).connect(
                remoteAnalyser,
            );
        }
    };

    if (localStream) {
        localStream.getTracks().forEach((track) =>
            newPc.addTrack(track, localStream)
        );
    }
}

// Надежное добавление ICE кандидатов (решение Race Condition)
async function addIceCandidate(candidate) {
    // [FIX] pc может быть уже сброшен — не падаем.
    // [GLARE] В состоянии have-local-offer (наш оффер в полёте) кандидаты
    // применять нельзя — они относятся к другому поколению SDP. Кладём их
    // в очередь: flushIceQueue() применит после завершения negotiation.
    if (pc && pc.remoteDescription && pc.signalingState !== "have-local-offer") {
        try {
            await pc.addIceCandidate(new RTCIceCandidate(candidate));
        } catch (e) {
            // После rollback/ICE-restart могут прилетать кандидаты устаревшего
            // поколения — это ожидаемо, не роняем обработчик.
            console.warn("[ICE] Кандидат отклонён:", e?.message || e);
        }
    } else {
        iceQueue.push(candidate);
    }
}

async function flushIceQueue() {
    // [FIX] Проверяем pc на каждой итерации: звонок могли сбросить,
    // пока мы добавляли предыдущего кандидата.
    while (iceQueue.length && pc) {
        try {
            await pc.addIceCandidate(new RTCIceCandidate(iceQueue.shift()));
        } catch (e) {
            console.warn("[ICE] Кандидат из очереди отклонён:", e?.message || e);
        }
    }
}

// ========== ЛОГИКА ЗВОНКОВ (Чистые функции действий) ==========

// Роутер кнопки
function onCallButtonClick() {
    switch (callState) {
        case "IDLE":
            startOutgoingCall();
            break;
        case "RINGING_IN":
            answerCall();
            break;
        case "AWAITING_OFFER":
            break; // Ждем, кнопка заблокирована
        case "RINGING_OUT":
        case "CONNECTING":
        case "CONNECTED":
            hangUp();
            break;
    }
}

async function startOutgoingCall() {
    const seq = ++callSeq; // [FIX RC-3] фиксируем поколение звонка
    try {
        // [FIX] Раньше signalingSocket.send() без проверки readyState кидал
        // TypeError и показывал "CONNECTION FAILED" вместо понятной причины.
        if (!signalingSocket || signalingSocket.readyState !== WebSocket.OPEN) {
            throw new Error("NO_SIGNALING");
        }

        updateCallUI("CALLING>>>", "btn-blue", "CONNECTING...", "#aaa", true);
        callState = "RINGING_OUT";

        await setupAudio();
        if (seq !== callSeq) return; // [FIX RC-3] звонок уже сброшен
        await createPeerConnection();
        if (seq !== callSeq) return;

        signalingSocket.send(
            JSON.stringify({ type: "call_request", target: currentCallTarget }),
        );
        startRingTimeout(); // [TIMEOUT] 60с на ответ, иначе NO ANSWER
        updateCallUI("CALLING>>>", "btn-blue", "RINGING>>>", "#aaa", true);
    } catch (err) {
        console.error("Call start error:", err);
        // [FIX] Определяем отказ в микрофоне по стандартному err.name,
        // а не по подстроке в сообщении (зависит от языка браузера).
        const micDenied =
            err?.name === "NotAllowedError" || err?.name === "NotFoundError";
        updateCallUI(
            "CALL",
            "btn-green",
            micDenied
                ? "MIC ACCESS DENIED"
                : (err?.message === "NO_SIGNALING"
                    ? "NO CONNECTION"
                    : "CONNECTION FAILED"),
            "#ff4a4a",
            false,
        );
        // [FIX] Раньше callState зависал в RINGING_OUT после ошибки
        callState = "IDLE";
        cleanupPeerConnection();
    }
}

async function answerCall() {
    const seq = ++callSeq; // [FIX RC-3]
    try {
        updateCallUI(
            "CONNECTING...",
            "btn-blue",
            "WAITING FOR OFFER",
            "#aaa",
            true,
        );
        callState = "AWAITING_OFFER";

        await fetchIceServers();
        if (seq !== callSeq) return; // [FIX RC-3]
        if (!localStream) await setupAudio();
        if (seq !== callSeq) return;
        if (!pc) await createPeerConnection();
        if (seq !== callSeq) return;

        if (signalingSocket?.readyState === WebSocket.OPEN) {
            signalingSocket.send(
                JSON.stringify({ type: "accept_call", target: currentCallTarget }),
            );
        }

        // [FIX RC-1] Если оффер уже пришел, пока мы готовили pc (несколько
        // await выше) — он лежит в pendingOfferSdp, забираем его.
        // Обратная гонка (оффер придет чуть позже) закрыта в case "offer":
        // там при AWAITING_OFFER и отсутствующем pc оффер тоже кладется
        // в pendingOfferSdp, а не валился в null pc.
        if (pendingOfferSdp) {
            const sdp = pendingOfferSdp;
            pendingOfferSdp = null;
            await processOfferAndAnswer(sdp, seq);
        }
    } catch (err) {
        console.error("Answer error:", err);
        pendingOfferSdp = null; // [FIX] не тащим битый оффер в следующую попытку
        updateCallUI("ANSWER", "btn-green", "ERROR", "#ff4a4a", false);
        callState = "RINGING_IN"; // Возврат в предыдущее состояние
    }
}

async function processOfferAndAnswer(sdp, seq = callSeq) {
    if (!pc) return; // [FIX] null-guard
    const myPc = pc;

    callState = "CONNECTING";
    updateCallUI("CONNECTING...", "btn-blue", "CONNECTING...", "#aaa", true);

    await myPc.setRemoteDescription(
        new RTCSessionDescription({ type: "offer", sdp }),
    );
    if (pc !== myPc || seq !== callSeq) return; // [FIX RC-3] сбросили, пока ждали

    await flushIceQueue(); // Apply queued candidates
    if (pc !== myPc || seq !== callSeq) return;

    const answer = await myPc.createAnswer();
    if (pc !== myPc || seq !== callSeq) return;
    await myPc.setLocalDescription(answer);
    if (pc !== myPc || seq !== callSeq) return;
    await flushIceQueue(); // [GLARE] кандидаты, накопленные за время negotiation
    if (pc !== myPc || seq !== callSeq) return;

    if (signalingSocket?.readyState === WebSocket.OPEN) {
        signalingSocket.send(JSON.stringify({
            type: "answer",
            target: currentCallTarget,
            sdp: myPc.localDescription.sdp,
        }));
    }

    onCallConnected();
}

function onCallConnected() {
    // [FIX RC-3] Если звонок уже сброшен (hangUp успел пройти мимо),
    // не "оживляем" его: не поднимаем callState и не запускаем таймер.
    if (callState !== "CONNECTING" && callState !== "RINGING_OUT") return;
    clearRingTimeout(); // [TIMEOUT] дозвонились — таймаут не нужен
    callState = "CONNECTED";
    updateCallUI("END CALL", "btn-red", "CONNECTED", "#4aff4a", false);
    startTimer();
}

function hangUp() {
    if (callState === "IDLE") return;

    // [FIX RC-3] Аннулируем все незавершённые async-цепочки этого звонка
    callSeq++;

    if (reconnectTimeout) {
        clearTimeout(reconnectTimeout);
        reconnectTimeout = null;
    }
    clearRingTimeout(); // [TIMEOUT]

    // Проверяем, что сокет не просто существует, но и ОТКРЫТ
    if (
        signalingSocket && signalingSocket.readyState === WebSocket.OPEN &&
        currentCallTarget
    ) {
        signalingSocket.send(
            JSON.stringify({
                type: "call_end",
                target: currentCallTarget,
                call_id: activeCallId, // [CALL_ID] сервер валидирует по своему реестру
            }),
        );
    } else {
        console.warn("[Signal] Не удалось отправить call_end — нет сети.");
    }

    callState = "IDLE";
    cleanupPeerConnection();

    updateCallUI("CALL", "btn-green", "ENDED", "#ff4a4a", false);
    stopTimer();
    stopCallSimulation();

    setTimeout(() => {
        if (currentUIState === "CONTENT") renderConnectModule();
    }, 1000);
}

function cleanupPeerConnection() {
    if (pc) {
        pc.close();
        pc = null;
    }
    if (localStream) {
        localStream.getTracks().forEach((t) => t.stop());
        localStream = null;
    }
    if (remoteAudioElement) {
        remoteAudioElement.pause();
        remoteAudioElement.srcObject = null;
        remoteAudioElement = null;
    }
    if (audioContext && audioContext.state !== "closed") audioContext.close();

    audioContext = null;
    localAnalyser = null;
    localDataArray = null;
    remoteAnalyser = null;
    remoteDataArray = null;
    iceQueue = [];
    // [CALL_ID] ID звонка живёт ровно столько же, сколько соединение
    activeCallId = null;
    // [FIX RC-4] Обязательно чистим отложенный оффер. Раньше SDP от
    // ПРОШЛОГО (пропущенного) звонка доживал до следующего звонка, и
    // answerCall() строил соединение по устаревшему описанию.
    pendingOfferSdp = null;
}

// ========== СИГНАЛИНГ (WebSocket) ==========

function connectSignaling(token) {
    if (pingInterval) {
        clearInterval(pingInterval);
        pingInterval = null;
    }

    // [FIX RC-5] Держим ЛОКАЛЬНУЮ ссылку на создаваемый сокет. В onclose
    // сравниваем signalingSocket с ней: если нас уже заменили новым сокетом —
    // это закрытие устаревшего, его игнорируем. Это лечит:
    //   - убийство ping-интервала НОВОГО сокета старым onclose;
    //   - двойной reconnect (два живых сокета одновременно).
    const socket = new WebSocket(
        `${WS_URL}/ws?token=${encodeURIComponent(token)}`,
    );
    signalingSocket = socket;

    socket.onopen = () => {
        if (signalingSocket !== socket) return; // уже не актуален
        console.log("[Signal] Connected as", token);

        pingInterval = setInterval(() => {
            if (signalingSocket === socket && socket.readyState === WebSocket.OPEN) {
                socket.send(JSON.stringify({ type: "ping" }));
            }
        }, 20000);
    };

    socket.onmessage = (event) => {
        if (signalingSocket !== socket) return; // сообщения старого сокета
        let msg;
        try {
            msg = JSON.parse(event.data);
        } catch (e) {
            console.warn("[Signal] Некорректный JSON:", e);
            return;
        }
        if (msg.type === "pong") return;

        // [FIX RC-2] Все сообщения обрабатываются строго последовательно
        signalingQueue = signalingQueue
            .then(() => handleSignalingMessage(msg))
            .catch((e) => console.error("[Signal] Ошибка обработки:", e));
    };

    socket.onclose = (event) => {
        // [FIX RC-5] Закрытие устаревшего сокета — нас уже заменили
        if (signalingSocket !== socket) return;

        clearInterval(pingInterval);
        pingInterval = null;
        console.log(
            `[Signal] Disconnected. Code ${event.code}, Reason: ${event.reason}`,
        );

        // [FIX RC-5] 4002 = сервер заменил нас новой сессией (второе
        // устройство / повторный логин). Автопереподключение здесь устроило
        // бы бесконечную "войну сокетов" между устройствами.
        if (event.code === 4002) {
            console.warn(
                "[Signal] Сессия заменена новым подключением (4002). Автопереподключение отключено.",
            );
            return;
        }

        if (event.code === 1006) {
            console.warn("proxy (1006)");
        }
        if (callState !== "IDLE" && callState !== "CONNECTED") {
            hangUp();
        } else if (callState === "CONNECTED") {
            const statusEl = document.getElementById("call-status");
            if (statusEl) {
                statusEl.innerText = "WEBSOCKET RECONNECTING";
                statusEl.style.color = "#ff9900";
            }
        }

        setTimeout(() => {
            if (signalingSocket !== socket) return; // уже переподключились
            const session = JSON.parse(
                localStorage.getItem("agora_session") || "{}",
            );
            if (session.token) connectSignaling(session.token);
        }, 3000);
    };

    socket.onerror = (err) => console.error("[Signal] Error", err);
}

async function handleSignalingMessage(msg) {
    switch (msg.type) {
        case "start_offer": {
            if (callState !== "RINGING_OUT") return;
            if (!pc) return; // [FIX] null-guard
            if (msg.call_id) activeCallId = msg.call_id; // [CALL_ID] резервная подстраховка
            try {
                const offer = await pc.createOffer();
                if (!pc || callState !== "RINGING_OUT") return; // [FIX RC-3]
                await pc.setLocalDescription(offer);
                if (!pc || callState !== "RINGING_OUT") return;
                if (signalingSocket?.readyState === WebSocket.OPEN) {
                    signalingSocket.send(
                        JSON.stringify({
                            type: "offer",
                            target: currentCallTarget,
                            sdp: pc.localDescription.sdp,
                        }),
                    );
                }
                updateCallUI(
                    "CALLING>>>",
                    "btn-blue",
                    "RINGING>>>",
                    "#aaa",
                    true,
                );
            } catch (e) {
                console.error("Offer creation failed:", e);
                hangUp();
            }
            break;
        }

        case "incoming_call": {
            if (callState !== "IDLE") return; // Защита: если уже звоним, игнорируем
            callSeq++; // [FIX RC-3] новое поколение — гасим хвосты старого
            currentCallTarget = msg.from;
            activeCallId = msg.call_id || null; // [CALL_ID]
            initCallInterface(msg.from);
            callState = "RINGING_IN";
            fetchIceServers(); // Предзагрузка ICE в фоне
            updateCallUI(
                "ANSWER",
                "btn-green",
                "INCOMING CALL",
                "#ff9900",
                false,
            );
            break;
        }

        case "call_created": {
            // [CALL_ID] Сервер зарегистрировал наш исходящий звонок и выдал ID.
            // Придёт сразу после call_request — раньше, чем любые offer/candidate.
            if (callState === "RINGING_OUT") {
                activeCallId = msg.call_id || null;
            }
            break;
        }

        case "busy": {
            // [BUSY] Абонент уже в звонке — сервер сказал сразу, не ждём 60с.
            if (callState !== "RINGING_OUT") return;
            console.log("[Call] Абонент занят");
            callSeq++; // гасим незавершённые async-цепочки
            clearRingTimeout();
            callState = "IDLE";
            cleanupPeerConnection();
            updateCallUI("CALL", "btn-green", "LINE BUSY", "#ff4a4a", false);
            stopTimer();
            stopCallSimulation();
            setTimeout(() => {
                if (currentUIState === "CONTENT") renderConnectModule();
            }, 1500);
            break;
        }

        case "offer": {
            // [CALL_ID] хвосты чужого/прошлого звонка — молча игнорируем
            if (isStaleCallMsg(msg)) return;

            // [FIX] Офферы не от текущего собеседника игнорируем,
            // кроме случая IDLE/RINGING_IN (тогда это новый входящий).
            if (
                msg.from && currentCallTarget &&
                msg.from !== currentCallTarget &&
                callState !== "IDLE" && callState !== "RINGING_IN"
            ) {
                console.warn("[Signal] Оффер от постороннего абонента, игнорируем:", msg.from);
                return;
            }

            if (callState === "CONNECTED") {
                // ЭТО ICE RESTART ОТ СОБЕСЕДНИКА (ОН СМЕНИЛ СЕТЬ)
                console.log(
                    "[ICE] Получен запрос на переподключение сети от собеседника!",
                );
                if (!pc) return; // [FIX] null-guard
                const myPc = pc; // [FIX RC-3]

                // [GLARE] ОБА одновременно послали оффер (мы только что
                // отправили свой restart-оффер, он ещё в полёте).
                // Решаем детерминированно, по алфавиту ролей — см. isImpolitePeer().
                if (myPc.signalingState === "have-local-offer") {
                    if (isImpolitePeer()) {
                        // Мы impolite: наш оффер главнее, входящий игнорируем.
                        // Своё состояние получит answer от собеседника.
                        console.log("[Glare] Мы impolite — игнорируем входящий оффер, ждём answer на свой");
                        return;
                    }
                    // Мы polite: откатываем свой оффер и принимаем их.
                    console.log("[Glare] Мы polite — откатываем свой оффер, принимаем входящий");
                    await myPc.setRemoteDescription(
                        new RTCSessionDescription({ type: "rollback" }),
                    );
                    if (pc !== myPc || callState !== "CONNECTED") return; // [FIX RC-3]
                }

                updateCallUI(
                    "END CALL",
                    "btn-vosklet",
                    "SYNC...",
                    "#ff9900",
                    false,
                );

                try {
                    await myPc.setRemoteDescription(
                        new RTCSessionDescription({ type: "offer", sdp: msg.sdp }),
                    );
                    if (pc !== myPc || callState !== "CONNECTED") return; // [FIX RC-3]
                    const answer = await myPc.createAnswer();
                    if (pc !== myPc || callState !== "CONNECTED") return;
                    await myPc.setLocalDescription(answer);
                    if (pc !== myPc || callState !== "CONNECTED") return;
                    await flushIceQueue(); // [GLARE] кандидаты, ждущие в очереди
                    if (pc !== myPc || callState !== "CONNECTED") return;

                    if (signalingSocket?.readyState === WebSocket.OPEN) {
                        signalingSocket.send(JSON.stringify({
                            type: "answer",
                            target: currentCallTarget,
                            sdp: myPc.localDescription.sdp,
                        }));
                    }

                    updateCallUI(
                        "END CALL",
                        "btn-red",
                        "CONNECTED",
                        "#4aff4a",
                        false,
                    );
                } catch (e) {
                    console.error("[ICE] Ошибка обработки restart-оффера:", e);
                }
            } else if (callState === "AWAITING_OFFER") {
                // [FIX RC-1] КЛЮЧЕВАЯ ТОЧКА: оффер пришел, пока answerCall()
                // еще внутри своих await (fetchIceServers/setupAudio/
                // createPeerConnection). Раньше здесь сразу звался
                // processOfferAndAnswer и падал на null pc — оффер терялся,
                // звонок зависал. Теперь кладем в pendingOfferSdp: answerCall()
                // заберет его сразу после создания pc.
                if (!pc) {
                    pendingOfferSdp = msg.sdp;
                } else {
                    await processOfferAndAnswer(msg.sdp);
                }
            } else if (callState === "IDLE" || callState === "RINGING_IN") {
                // Оффер пришел ДО того, как пользователь нажал ANSWER
                initCallInterface(msg.from);
                callState = "RINGING_IN";
                updateCallUI(
                    "ANSWER",
                    "btn-green",
                    "INCOMING CALL",
                    "#ff9900",
                    false,
                );

                // [CALL_ID] холодный старт из push: incoming_call мы не видели
                // (ушёл в мёртвый сокет), ID приходит вместе с оффером
                if (msg.call_id) activeCallId = msg.call_id;

                // Просто сохраняем SDP в переменную. Когда юзер нажмет кнопку,
                // функция answerCall() сама его подхватит.
                pendingOfferSdp = msg.sdp;
            }
            break;
        }

        case "answer": {
            // [CALL_ID] хвосты чужого звонка игнорируем
            if (isStaleCallMsg(msg)) return;

            // [FIX RESTART-ANSWER] Раньше здесь был жёсткий фильтр
            // "RINGING_OUT|CONNECTING" — из-за него answer на наш
            // ICE-restart (мы уже CONNECTED, signalingState = have-local-offer)
            // ВЫБРАСЫВАЛСЯ, и рестарт зависал навсегда. Теперь такой answer
            // легально применяется.
            const isRestartAnswer =
                callState === "CONNECTED" && pc &&
                pc.signalingState === "have-local-offer";
            if (
                callState !== "RINGING_OUT" && callState !== "CONNECTING" &&
                !isRestartAnswer
            ) {
                return;
            }
            if (!pc) return; // [FIX] null-guard
            try {
                await pc.setRemoteDescription(
                    new RTCSessionDescription({ type: "answer", sdp: msg.sdp }),
                );
                if (!pc) return; // [FIX RC-3] сбросили, пока ждали
                await flushIceQueue();
                if (isRestartAnswer) {
                    // Это ответ на наш ICE-restart — звонок уже подключён,
                    // таймер идёт. Просто восстанавливаем UI после SYNC...
                    updateCallUI(
                        "END CALL",
                        "btn-red",
                        "CONNECTED",
                        "#4aff4a",
                        false,
                    );
                    if (reconnectTimeout) {
                        clearTimeout(reconnectTimeout);
                        reconnectTimeout = null;
                    }
                } else {
                    onCallConnected();
                }
            } catch (e) {
                console.error("[Signal] Не удалось применить answer:", e);
                hangUp();
            }
            break;
        }

        case "candidate": {
            // Игнорируем запоздалые пакеты от старых сессий, если мы уже в IDLE
            if (callState === "IDLE") {
                console.log(
                    "[Signal] Проигнорирован запоздалый кандидат (мы уже оффлайн)",
                );
                return;
            }
            // [CALL_ID] хвосты чужого звонка игнорируем
            if (isStaleCallMsg(msg)) return;
            // [FIX] Кандидат не от текущего собеседника — мусор прошлой сессии
            if (msg.from && currentCallTarget && msg.from !== currentCallTarget) {
                return;
            }
            if (pc) await addIceCandidate(msg.candidate);
            break;
        }

        case "peer_disconnected": {
            // [CALL_ID] сброс чужого/прошлого звонка не должен ронять текущий
            if (isStaleCallMsg(msg)) return;
            hangUp();
            break;
        }
    }
}

function startWaveAnimation() {
    const ctxRemote = document.getElementById("remote-canvas")?.getContext(
        "2d",
    );
    const ctxMic = document.getElementById("mic-canvas")?.getContext(
        "2d",
    );
    if (!ctxRemote || !ctxMic) return;

    function draw(ctx, color, volumeData) {
        const w = ctx.canvas.width;
        const h = ctx.canvas.height;
        ctx.clearRect(0, 0, w, h);
        ctx.fillStyle = color;

        const bars = 80;
        const barWidth = w / bars;
        for (let i = 0; i < bars; i++) {
            const val = (callState === "CONNECTED") ? (volumeData[i] || 0) : 0;
            const barHeight = (val / 255) * h * 0.9 + 2;
            ctx.fillRect(
                i * barWidth,
                (h - barHeight) / 2,
                barWidth - 2,
                barHeight,
            );
        }
    }

    function loop() {
        if (currentUIState !== "CONTENT") {
            animationFrameId = null;
            return;
        }

        if (localAnalyser && !isMuted && localDataArray) {
            localAnalyser.getByteFrequencyData(localDataArray);
        } else if (localDataArray) {
            localDataArray.fill(0);
        }

        if (remoteAnalyser && remoteDataArray) {
            remoteAnalyser.getByteFrequencyData(remoteDataArray);
        } else if (remoteDataArray) {
            remoteDataArray.fill(0);
        }

        draw(ctxRemote, "#a0a0a0", remoteDataArray || []);
        draw(ctxMic, isMuted ? "#552222" : "#a0a0a0", localDataArray);

        animationFrameId = requestAnimationFrame(loop);
    }
    loop();
}

# Заменяем старый ConnectionManager на UserManager
class UserManager:
    def __init__(self):
        # user_id (role) → WebSocket
        self.online: dict[str, WebSocket] = {}

    async def connect_user(self, user_id: str, ws: WebSocket):
        # Если уже был онлайн с другого устройства — отключаем старый
        old_ws = self.online.get(user_id)
        if old_ws is not None and old_ws is not ws:
            try:
                await old_ws.close(code=4002, reason="replaced_by_new_session")
            except Exception:
                pass
        await ws.accept()
        self.online[user_id] = ws

    # [FIX RC-6] ГЛАВНАЯ СЕРВЕРНАЯ ГОНКА (race condition):
    # раньше disconnect_user(user_id) безусловно делал pop(user_id).
    # Сценарий: юзер переподключился → connect_user закрыл старый ws и
    # записал новый → finally старого хендлера сработал ПОЗЖЕ и выкинул
    # из словаря НОВЫЙ сокет → юзер числился оффлайн, звонки уходили
    # в push вместо WS. Теперь удаляем запись только если в словаре лежит
    # именно закрывающийся сокет.
    def disconnect_user(self, user_id: str, ws: WebSocket):
        if self.online.get(user_id) is ws:
            self.online.pop(user_id, None)

    def get_user_ws(self, user_id: str) -> WebSocket | None:
        return self.online.get(user_id)

# [CALL REGISTRY] Реестр активных звонков (в памяти).
# call_id генерирует СЕРВЕР — клиентскому ID доверять нельзя.
# Отсюда: занятость (busy), call_id в сообщениях, уведомление партнёра
# при закрытии вкладки, и защита от «хвостов» старого звонка.
# Оговорка: при рестарте процесса (Render free tier засыпает) реестр
# теряется — медиа (ICE) продолжит идти напрямую, а вот мгновенное
# уведомление о закрытии вкладки для тех звонков уже не придёт.
class CallRegistry:
    def __init__(self):
        # call_id → {"caller", "callee", "state": RINGING|ACTIVE, "ts"}
        self.calls: dict[str, dict] = {}
        # user_id → call_id (у юзера один звонок одновременно)
        self.user_calls: dict[str, str] = {}

    def _purge(self, call_id: str) -> dict | None:
        call = self.calls.pop(call_id, None)
        if call:
            # Чистим индекс только если он указывает на ЭТОТ звонок
            # (защита от выкидывания более свежей записи — та же гонка,
            # что и в UserManager.disconnect_user).
            if self.user_calls.get(call["caller"]) == call_id:
                self.user_calls.pop(call["caller"], None)
            if self.user_calls.get(call["callee"]) == call_id:
                self.user_calls.pop(call["callee"], None)
        return call

    def cleanup_stale(self, ttl: float = 120.0):
        """Убирает зависшие RINGING-звонки (звонили, никто не взял,
        call_end потерялся). ACTIVE не трогаем — их закрывает
        WS-finally при разрыве."""
        now = time.time()
        for call_id in list(self.calls.keys()):
            call = self.calls[call_id]
            if call["state"] == "RINGING" and now - call["ts"] > ttl:
                self._purge(call_id)

    def call_of(self, user_id: str) -> tuple[str | None, dict | None]:
        """Звонок, в котором участвует юзер: (call_id, call) или (None, None)."""
        call_id = self.user_calls.get(user_id)
        if call_id is None:
            return None, None
        call = self.calls.get(call_id)
        if call is None:
            # Битый индекс — чиним на месте
            self.user_calls.pop(user_id, None)
            return None, None
        return call_id, call

    def create(self, caller: str, callee: str) -> str:
        call_id = uuid.uuid4().hex[:12]
        self.calls[call_id] = {
            "caller": caller,
            "callee": callee,
            "state": "RINGING",
            "ts": time.time(),
        }
        self.user_calls[caller] = call_id
        self.user_calls[callee] = call_id
        return call_id

    def activate(self, call_id: str):
        call = self.calls.get(call_id)
        if call:
            call["state"] = "ACTIVE"
            call["ts"] = time.time()

    def end_for_user(self, user_id: str) -> tuple[str, str] | None:
        """Заканчивает звонок юзера. Возвращает (call_id, partner_id)
        или None, если юзер не в звонке."""
        call_id, call = self.call_of(user_id)
        if not call:
            return None
        self._purge(call_id)
        partner = call["callee"] if call["caller"] == user_id else call["caller"]
        return call_id, partner

    user_info = get_user_from_token(token)
    if not user_info:
        raise HTTPException(status_code=401, detail="Invalid token")

    try:
        print(f"[Push] Сохраняю подписку для user_id={user_info['role']}")
        res = supabase.table("push_subs").upsert(
            {
                "user_id": user_info["role"],
                "sub_data": data.subscription,
            },
            on_conflict="user_id",
        ).execute()

        if hasattr(res, 'error') and res.error:
            raise Exception(f"Supabase error: {res.error}")
        print("[Push] Успешно сохранено")
        return {"status": "ok"}

    except Exception as e:
        print(f"[Push] Первая попытка не удалась ({e}), пробую upsert без on_conflict")
        try:
            supabase.table("push_subs").delete().eq("user_id", user_info["role"]).execute()
            res = supabase.table("push_subs").insert({
                "user_id": user_info["role"],
                "sub_data": data.subscription,
            }).execute()
            if hasattr(res, 'error') and res.error:
                raise Exception(f"Supabase insert error: {res.error}")
            print("[Push] Успешно сохранено (замена)")
            return {"status": "ok"}
        except Exception as inner_e:
            print(f"[Push] Полная ошибка: {inner_e}")
            import traceback
            traceback.print_exc()
            raise HTTPException(status_code=500, detail=f"Push subscription failed: {inner_e}")

@app.post("/push/unsubscribe")
async def unsubscribe_push(data: dict, token: str = Query(...)):
    user_info = get_user_from_token(token)
    if not user_info:
        raise HTTPException(status_code=401)

    endpoint = data.get("endpoint")
    if not endpoint:
        raise HTTPException(status_code=400, detail="Missing endpoint")

    supabase.table("push_subs") \
        .delete() \
        .eq("sub_data->>endpoint", endpoint) \
        .execute()

    return {"status": "ok"}

# Маршрутизация сообщений
async def handle_signal_message(sender_id: str, message: dict, sender_ws: WebSocket):
    msg_type = message.get("type")

    # Подчищаем зависшие RINGING-звонки при каждом сообщении
    # (реестр крошечный, итерация бесплатна)
    call_registry.cleanup_stale()

    if msg_type == "call_request":
        # [FIX] message["target"] при отсутствии поля кидал KeyError,
        # который обрывал websocket ЗВОНЯЩЕГО (общий except в цикле).
        target = message.get("target")
        if not target or target == sender_id:
            return

        # [BUSY] Цель уже в звонке (RINGING или ACTIVE) — сразу отказ,
        # звонящий не висит 60 секунд в RINGING.
        _, target_call = call_registry.call_of(target)
        if target_call:
            print(f"[Call] {target} занят — отказ {sender_id}")
            await _safe_send(sender_ws, {"type": "busy"})
            return

        # Сам звонящий завис в старом звонке (retry после таймаута и т.п.)
        # — прибиваем старый, партнёра уведомляем.
        stale = call_registry.end_for_user(sender_id)
        if stale:
            stale_id, stale_partner = stale
            stale_ws = user_manager.get_user_ws(stale_partner)
            if stale_ws:
                await _safe_send(stale_ws, {
                    "type": "peer_disconnected",
                    "call_id": stale_id,
                })

        # [CALL_ID] Регистрируем звонок ДО доставки: без записи в реестре
        # accept_call из push-холодного старта не нашёл бы, что активировать.
        call_id = call_registry.create(sender_id, target)

        target_ws = user_manager.get_user_ws(target)
        delivered = False
        if target_ws is not None:
            delivered = await _safe_send(target_ws, {
                "type": "incoming_call",
                "from": sender_id,
                "call_id": call_id,
                "room_id": f"{sender_id}_{target}"   # для совместимости, если нужно
            })
            if not delivered:
                # [FIX] Протухший сокет в словаре — чистим и считаем цель оффлайн
                user_manager.disconnect_user(target, target_ws)

        if delivered:
            # [CALL_ID] Сообщаем ЗВОНЯЩЕМУ его же call_id — теперь он тоже
            # может отфильтровывать хвосты чужих звонков.
            await _safe_send(sender_ws, {
                "type": "call_created",
                "call_id": call_id,
                "target": target,
            })
        else:
            # Здесь отправка push (звонок остаётся RINGING в реестре —
            # TTL 120с или call_end его уберут)
            print(f"[PUSH] {target} оффлайн. Ищем подписку в БД...")
            response = supabase.table("push_subs").select("sub_data").eq("user_id", target).execute()

            if response.data:
                for row in response.data:
                    sub_data = row["sub_data"]
                    payload = json.dumps({
                        "title": "Входящий вызов",
                        "body": f"{sender_id} вызывает вас...",
                        "caller": sender_id,
                        "target": target,
                        "type": "INCOMING_CALL"
                    })
                    headers = {
                        "Urgency":"high",
                        "Topic":"incoming-calls",
                        "apns-priority": "10",
                        "apns-push-type": "alert",
                        "apns-expiration": "45"
                    }
                    try:
                        await asyncio.to_thread(webpush,
                            subscription_info=sub_data,
                            data=payload,
                            vapid_private_key=VAPID_PRIVATE_KEY,
                            vapid_claims=VAPID_CLAIMS,
                            ttl=45,
                            headers=headers
                        )
                        print(f"[PUSH] Уведомление отправлено {target}")
                    except WebPushException as e:
                        print(f"[PUSH] Ошибка отправки: {e}")
                        # [FIX] Мёртвая подписка (410/404) — удаляем из БД,
                        # иначе она будет вечно висеть и валить отправку.
                        status = None
                        if getattr(e, "response", None) is not None:
                            status = getattr(e.response, "status_code", None)
                        if status in (404, 410):
                            try:
                                dead_endpoint = (sub_data or {}).get("endpoint")
                                if dead_endpoint:
                                    supabase.table("push_subs") \
                                        .delete() \
                                        .eq("sub_data->>endpoint", dead_endpoint) \
                                        .execute()
                            except Exception as del_e:
                                print(f"[PUSH] Не удалось удалить мёртвую подписку: {del_e}")
            else:
                # Абонент вообще недоступен — откатываем регистрацию звонка
                call_registry._purge(call_id)
                await _safe_send(sender_ws, {"type": "peer_disconnected"})

    elif msg_type == "accept_call":
        target = message.get("target")
        call_id, call = call_registry.call_of(sender_id)

        # [CALL_ID] Звонок истёк (TTL) или реестр потерян при рестарте —
        # не даём принимающему висеть в AWAITING_OFFER.
        if (
            not call or
            call["callee"] != sender_id or
            (target and call["caller"] != target)
        ):
            await _safe_send(sender_ws, {"type": "peer_disconnected"})
            return

        # Защита от двойного нажатия ANSWER / дубликата accept_call
        if call["state"] == "ACTIVE":
            return

        call_registry.activate(call_id)

        caller_ws = user_manager.get_user_ws(call["caller"])
        if caller_ws:
            ok = await _safe_send(caller_ws, {
                "type": "start_offer",
                "target": sender_id,   # тому, кто звонил, говорим начинать
                "call_id": call_id,
            })
            if not ok:
                # Звонящий исчез, пока звонок звонел
                call_registry._purge(call_id)
                await _safe_send(sender_ws, {"type": "peer_disconnected"})
        else:
            call_registry._purge(call_id)
            await _safe_send(sender_ws, {"type": "peer_disconnected"})

    elif msg_type in ("offer", "answer", "candidate"):
        target = message.get("target")
        if target:
            target_ws = user_manager.get_user_ws(target)
            if target_ws:
                # Добавляем поле from для идентификации отправителя
                forwarded = {**message, "from": sender_id}
                # [CALL_ID] Штампуем из реестра (сервер — источник истины).
                # Нет записи (рестарт сервера) — релеим без call_id,
                # клиент тогда фильтрует по from (легаси-путь).
                call_id, _call = call_registry.call_of(sender_id)
                if call_id:
                    forwarded["call_id"] = call_id
                await _safe_send(target_ws, forwarded)

    elif msg_type == "call_end":
        # [CALL_ID] call_end работает ТОЛЬКО через реестр: сервер сам знает,
        # кому слать peer_disconnected. Клиентский call_id не используется —
        # доверять ему нельзя.
        ended = call_registry.end_for_user(sender_id)
        if ended:
            call_id, partner = ended
            partner_ws = user_manager.get_user_ws(partner)
            if partner_ws:
                await _safe_send(partner_ws, {
                    "type": "peer_disconnected",
                    "call_id": call_id,
                })
        # Если записи нет (рестарт) — сознательно НИЧЕГО не делаем:
        # слепой релей по target мог бы убить ДЕЙСТВУЮЩИЙ звонок
        # другого юзера с таким же именем.

@app.post("/call/decline")
async def decline_call(data: DeclineCallRequest):
    # Отклонение из push-уведомления. Ищем звонок между этими двумя
    # участниками в реестре (кто из них звонивший — проверяем по полям).
    call_id = None
    reg_id, call = call_registry.call_of(data.target)
    if call and call["caller"] == data.caller and call["callee"] == data.target:
        call_id = reg_id
        call_registry._purge(reg_id)
    else:
        # Fallback: реестр пуст (рестарт) — пробуем по звонящему напрямую
        reg_id2, call2 = call_registry.call_of(data.caller)
        if call2 and call2["callee"] == data.target:
            call_id = reg_id2
            call_registry._purge(reg_id2)

    caller_ws = user_manager.get_user_ws(data.caller)
    if caller_ws:
        payload = {"type": "peer_disconnected"}
        if call_id:
            payload["call_id"] = call_id
        await _safe_send(caller_ws, payload)
        print(f"[Call] Звонок от {data.caller} был отклонен пользователем {data.target} через Push")
    return {"status": "ok"}

@app.websocket("/ws")
async def websocket_call(websocket: WebSocket, token: str = Query("guest")):
    user_info = get_user_from_token(token)
    if not user_info:
        await websocket.close(code=4001, reason="Invalid token")
        return

    user_id = user_info["role"]
    await user_manager.connect_user(user_id, websocket)

    try:
        while True:
            # Ждем сообщение от клиента.
            # Если тишина больше 40 секунд — считаем клиента мертвым
            try:
                data = await asyncio.wait_for(websocket.receive_json(), timeout=40.0)
            except asyncio.TimeoutError:
                print(f"[WS] Таймаут пинга от {user_id}. Убиваем зомби-соединение.")
                break

            # Обработка пинга-понга (НЕ пускаем это в основную логику звонков)
            if data.get("type") == "ping":
                await websocket.send_json({"type": "pong"})
                continue

            # Если это не пинг — отправляем в основную маршрутизацию
            await handle_signal_message(user_id, data, websocket)

    except WebSocketDisconnect:
        # Клиент отключился нормально (например, закрыл вкладку)
        pass
    except Exception as e:
        # Случайная ошибка (например, отправили кривой JSON)
        print(f"[WS] Ошибка у {user_id}: {e}")
    finally:
        # [FIX RC-6] Передаём сам сокет: запись будет удалена только если
        # в словаре всё ещё лежит ОН (а не его более свежая замена).
        user_manager.disconnect_user(user_id, websocket)

        # [TAB-CLOSE] Юзер закрыл вкладку / убил приложение / потерял сеть
        # во время звонка — мгновенно сообщаем партнёру. Раньше партнёр
        # звонил в пустоту до собственного таймаута.
        ended = call_registry.end_for_user(user_id)
        if ended:
            call_id, partner = ended
            print(f"[WS] {user_id} оборвал соединение посреди звонка — оповещаем {partner}")
            partner_ws = user_manager.get_user_ws(partner)
            if partner_ws:
                await _safe_send(partner_ws, {
                    "type": "peer_disconnected",
                    "call_id": call_id,
                })

const CACHE_NAME = "agora-hub-v59"; // [v59] glare + call_id + busy + tab-close — версия поднята, клиенты заберут новый script1.js
const ASSETS = [
  "./",
  "./index.html",
  "./script1.js",
  "./style1.css",
  "./logo.png",
  "./manifest.json",
];
const API_URL = "https://agora-service.onrender.com";

// 1. При установке кэшируем основные файлы (App Shell)
// [FIX] Раньше стоял cache.addAll(ASSETS) — он АТОМАРНЫЙ: если хоть один
// файл отдавался 404/500 (например, забыли залить logo.png), весь install
// падал, новый Service Worker никогда не активировался и приложение
// навсегда оставалось на старой версии. Кэшируем каждый файл отдельно:
// один битый файл больше не блокирует обновление.
self.addEventListener("install", (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) =>
      Promise.allSettled(ASSETS.map((url) => cache.add(url)))
    ),
  );
  self.skipWaiting();
});

// 2. реакция на запрос
self.addEventListener("message", (event) => {
  if (event.data === "GET_CACHE_NAME") {
    self.clients.matchAll().then((clients) => {
      clients.forEach((client) => {
        client.postMessage({ type: "CACHE_NAME", name: CACHE_NAME });
      });
    });
  }
});

// Удаление старых кешей
self.addEventListener("activate", (event) => {
  const cacheWhitelist = [CACHE_NAME];
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cacheName) => {
          if (cacheWhitelist.indexOf(cacheName) === -1) {
            console.log("[SW] Удаляем старый кеш:", cacheName);
            return caches.delete(cacheName);
          }
        }),
      );
    }),
  );
  return self.clients.claim();
});

self.addEventListener("fetch", (event) => {
  if (event.request.method !== "GET") return;
  event.respondWith(
    caches.match(event.request).then((cachedResponse) => {
      if (cachedResponse) return cachedResponse;
      return fetch(event.request).then((networkResponse) => {
        if (
          !networkResponse || networkResponse.status !== 200 ||
          networkResponse.type !== "basic"
        ) {
          return networkResponse;
        }
        const responseToCache = networkResponse.clone();
        caches.open(CACHE_NAME).then((cache) => {
          cache.put(event.request, responseToCache);
        });
        return networkResponse;
      });
    }),
  );
});

// Ловим Push от сервера
self.addEventListener("push", function (event) {
  if (!event.data) return;

  const payload = event.data.json();

  if (payload.type === "INCOMING_CALL") {
    const options = {
      body: payload.body,
      icon: "./icons/icon-192.png",
      badge: "./icons/icon-192.png",
      vibrate: [200, 100, 200, 100, 200, 100, 200],
      requireInteraction: true, // Пуш не исчезнет сам
      silent: false,
      timestamp: Date.now(),
      data: { caller: payload.caller, target: payload.target },
      // Убрали кнопку "Принять". Оставляем только "Отклонить".
      actions: [
        { action: "decline", title: "🔴 Отклонить" },
      ],
    };

    event.waitUntil(
      self.registration.showNotification(payload.title, options),
    );
  }
});

self.addEventListener("notificationclick", function (event) {
  event.notification.close();

  const action = event.action;
  const payloadData = event.notification.data || {};
  const caller = payloadData.caller || "unknown";
  const target = payloadData.target || "unknown";

  console.log("[SW] notificationclick:", { action, caller, target });

  // 🔴 1. Нажали кнопку "ОТКЛОНИТЬ"
  if (action === "decline") {
    event.waitUntil(
      self.clients.matchAll({ type: "window", includeUncontrolled: true }).then(
        function (clients) {
          let notified = false;

          // Если приложение свернуто - кидаем ему команду на сброс
          clients.forEach(function (client) {
            if (client.url.includes(self.registration.scope)) {
              client.postMessage({ type: "CALL_DECLINED", caller: caller });
              notified = true;
            }
          });

          // Если приложение вообще закрыто - шлем запрос на сервер
          if (!notified) {
            return fetch(`${API_URL}/call/decline`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ caller: caller, target: target }),
            }).catch((err) =>
              console.error("[SW] Ошибка отправки decline:", err)
            );
          }
        },
      ),
    );
    return;
  }

  // 🟢 2. Нажали НА ТЕЛО УВЕДОМЛЕНИЯ (action пустой) -> открываем/разворачиваем звонок
  if (!action) {
    const targetUrl = self.registration.scope + "?call=" +
      encodeURIComponent(caller);
    event.waitUntil(
      self.clients.matchAll({ type: "window", includeUncontrolled: true })
        .then(
          function (clientsList) {
            // Пытаемся найти уже открытую вкладку
            for (let i = 0; i < clientsList.length; i++) {
              const client = clientsList[i];
              if (
                client.url.includes(self.registration.scope) &&
                "focus" in client
              ) {
                return client.focus().then(function (focusedClient) {
                  if (focusedClient) {
                    focusedClient.postMessage({
                      type: "WAKE_UP_CALL",
                      caller: caller,
                    });
                  }
                });
              }
            }

            // Если вкладки нет (убита) -> Открываем приложение с нуля
            if (self.clients.openWindow) {
              return self.clients.openWindow(targetUrl);
            }
          },
        ),
    );
  }
});
